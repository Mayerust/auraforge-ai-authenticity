import requests

url = "https://api.auraforge.dev/v1/analyze"
headers = {"Authorization": "Bearer YOUR_API_KEY"}

with open("audio_sample.mp3", "rb") as f:
    r = requests.post(url, headers=headers, files={"file": f})

result = r.json()
print(f"Decision: {result['decision']}")
print(f"Probability: {result['ai_probability']}")
const analyzeAudio = async (file) => {
  const form = new FormData();
  form.append("file", file);

  const res = await fetch("https://api.auraforge.dev/v1/analyze", {
    method: "POST",
    headers: { "Authorization": "Bearer YOUR_API_KEY" },
    body: form
  });

  const { ai_probability, decision } = await res.json();

  if (decision === "BLOCK") {
    console.warn("Action required: Asset rejected.");
  }
};
